	/* breakdown: the vehicle stops where it is for an exponential repair time; failures come exponentially (mean MTBF) in calendar time.
	   {{START}} is when the repair starts: ac after a leg (a failure during the leg shows at its end), the failure time itself at boarding
	   (a failure while the vehicle stood idle is repaired from then on, so only the rest of the repair holds the load) */
	while V_vNextFail(A_vid) <= ac do begin
		set A_rep to rn RS_fail e {{MTTR}}
		if {{START}} >= V_warmup then inc V_vDown(A_vid) by A_rep
		set A_rep to {{START}} + A_rep
		set V_vNextFail(A_vid) to A_rep + rn RS_fail e {{MTBF}}
		if A_rep > ac then begin
			set A_veh color to red
			wait for A_rep - ac sec
			set A_veh color to inherit
		end
	end
