/* material {{M}} = {{NAME}} (group {{G}}) */
/* one consumer cycle per takt: call check, then consume or stay stopped; a stopped cycle re-checks every takt */
begin P_consume_{{M}} arriving procedure
	while 1 = 1 do begin
{{CALL}}
		if V_stock({{M}}) < V_need({{M}}) - 0.000001 then begin
			if V_stopped({{M}}) = 0 then begin
				set V_stopped({{M}}) to 1
				set V_stopSince({{M}}) to ac
				if V_events = 1 and ac >= V_warmup then print ac "\tstockout\t{{NAME}}\t\t0" to V_ev
			end
		end
		else begin
			if V_stopped({{M}}) = 1 then begin
				set V_stopped({{M}}) to 0
				if V_stopSince({{M}}) < V_warmup then set V_stopSince({{M}}) to V_warmup
				if ac > V_stopSince({{M}}) then begin
					inc V_stop({{M}}) by ac - V_stopSince({{M}})
					if V_events = 1 then print ac "\tstockin\t{{NAME}}\t\t" ac - V_stopSince({{M}}) to V_ev
				end
			end
{{CONSUME}}
		end
		if ac >= V_warmup then begin
			inc V_samples({{M}}) by 1
			tabulate V_stock({{M}}) in T_stock_{{M}}
			inc V_stockSum({{M}}) by V_stock({{M}})
			if V_stock({{M}}) < V_minStock({{M}}) then set V_minStock({{M}}) to V_stock({{M}})
			/* supply wait (value9..11): containers on vehicles held for the unload threshold, and those vehicles */
			inc V_waitSum({{M}}) by V_waitQty({{M}})
			inc V_waitVehSum({{M}}) by V_waitVeh({{M}})
			if V_waitQty({{M}}) > V_waitMax({{M}}) then set V_waitMax({{M}}) to V_waitQty({{M}})
		end
		wait for V_takt({{M}}) sec
	end
end
