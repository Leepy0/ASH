/* material {{M}} = {{NAME}} (group {{G}}) */
/* one consumer cycle per takt: call check, then consume or stay stopped; a stopped cycle re-checks every takt */
begin P_consume_{{M}} arriving procedure
	while 1 = 1 do begin
		/* reorder on the stock position (on hand + containers already called): count the containers short, as many as needed at once;
		   also call when the position cannot cover one cycle, or a call threshold below the per-cycle need would stop forever */
		set A_q to 0
		while V_stock({{M}}) + V_pending({{M}}) + A_q <= V_call({{M}}) + 0.000001 or V_stock({{M}}) + V_pending({{M}}) + A_q < V_need({{M}}) - 0.000001 do begin
			inc A_q by 1
		end
		/* one call per trip: a vehicle carries up to its Capacity containers (the clone keeps A_qty) */
		while A_q > 0 do begin
			if A_q > {{CAPACITY}} then set A_qty to {{CAPACITY}}
			else set A_qty to A_q
			dec A_q by A_qty
			inc V_pending({{M}}) by A_qty
			set V_extMat to {{M}}
			set V_tmp to F_ext_on_call()
			clone 1 load to P_supply_{{M}}
		end
		if V_stock({{M}}) < V_need({{M}}) - 0.000001 then begin
			if V_stopped({{M}}) = 0 then begin
				set V_stopped({{M}}) to 1
				set V_stopSince({{M}}) to ac
			end
		end
		else begin
			if V_stopped({{M}}) = 1 then begin
				set V_stopped({{M}}) to 0
				if V_stopSince({{M}}) < V_warmup then set V_stopSince({{M}}) to V_warmup
				if ac > V_stopSince({{M}}) then inc V_stop({{M}}) by ac - V_stopSince({{M}})
			end
			dec V_stock({{M}}) by V_need({{M}})
			if ac >= V_warmup then inc V_consumed({{M}}) by V_need({{M}})
		end
		if ac >= V_warmup then begin
			inc V_samples({{M}}) by 1
			tabulate V_stock({{M}}) in T_stock_{{M}}
			inc V_stockSum({{M}}) by V_stock({{M}})
			if V_stock({{M}}) < V_minStock({{M}}) then set V_minStock({{M}}) to V_stock({{M}})
		end
		wait for V_takt({{M}}) sec
	end
end

/* One supply job, start to end, driven by the load: it enters the system at the group's standby CP, where calls queue first come first served
   whether or not a vehicle is there, boards the first free vehicle, rides along every leg with travel to, and brings the vehicle home before it
   dies, so the park list never decides where a vehicle goes. AutoMod's vehicle states no longer mean anything (the load is on board the whole
   job), so busy time, distance and lead times are kept here. */
begin P_supply_{{M}} arriving procedure
	set A_grp to {{G}}
{{SET_TYPE}}
	set A_tCall to ac
	set V_extMat to {{M}}
	set V_tmp to F_ext_select_vehicle()
	move into {{SYS}}:{{HOME}}
	set A_veh to this load vehicle
	set A_vid to A_veh A_index + {{VID_OFFSET}}
	set A_tPick to ac
	/* a vehicle that came over from elsewhere to take this load: AutoMod counts that run as retrieving; the share since its last pickup is this job's */
	set V_x to (A_veh percent time retrieving) / 100 * ac
	set A_ret to V_x - A_veh A_rLast
	set A_veh A_rLast to V_x
{{FAIL_IDLE}}
{{EMPTY_SPEED}}
	travel to {{SYS}}:{{FROM}}
{{FAIL_LEG}}
{{LOADED_SPEED}}
	if ac >= V_warmup then tabulate ac - A_tCall in T_pick_{{M}}
	wait for {{LOAD_TIME}} sec
{{SUPPLY_LEGS}}
	set V_extMat to {{M}}
	set V_tmp to F_ext_before_unload()
	wait for {{UNLOAD_TIME}} sec
	inc V_stock({{M}}) by A_qty
	if ac >= V_warmup then begin
		inc V_supplies({{M}}) by A_qty
		tabulate ac - A_tCall in T_deliv_{{M}}
	end
	dec V_pending({{M}}) by A_qty
{{EMPTY_TYPE}}
{{AFTER_UNLOAD_SPEED}}
{{RETURN_LEGS}}
{{FINISH}}
	/* home first, always: the next job starts from the standby CP */
	travel to {{SYS}}:{{HOME}}
{{FAIL_LEG}}
{{FINISH_AT_HOME}}
{{SPEC_SPEED}}
	if A_tPick >= V_warmup then begin
		inc V_vBusy(A_vid) by A_ret + A_fin - A_tPick
		inc V_vJobs(A_vid) by 1
		set V_x to A_veh total distance traveled
		inc V_vDist(A_vid) by V_x - A_veh A_dLast
	end
	set A_veh A_dLast to A_veh total distance traveled
	send to die
end
