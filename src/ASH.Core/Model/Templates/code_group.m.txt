/* group {{G}} = AGVS vehicle type {{TYPE}} (named after the group): vehicles numbered 1..n; index > active parks for the whole run */
begin {{SYS}}.{{TYPE}} vehicle initialization procedure
	inc V_vehCount({{G}}) by 1
	set theVehicle A_index to V_vehCount({{G}})
	/* the model-wide vehicle number and the group: the deadlock watchdog (code_dlock.m) and the block claimed function (code_tsplit) */
	set theVehicle A_vid to V_vehCount({{G}}) + {{VID_OFFSET}}
	set theVehicle A_grp to {{G}}
	set V_dlVeh(theVehicle A_vid) to theVehicle
	/* stop {{STOP_M}} m before the vehicle ahead (feet, as the syntax help writes it): with a defined stop distance AutoMod works the brake distance out from speed and deceleration */
	set theVehicle defined stop distance to {{STOP_FT}} ft
	return true
end

begin {{SYS}}.{{TYPE}} task search procedure
	if A_index > V_active({{G}}) then begin
		inc V_parked({{G}}) by 1
		move into Q_park
		wait to be ordered on OL_forever
	end
end

/* AM scheduling lists are per location, so these type filters keep each group on its own materials and standby CPs */
begin {{SYS}}.{{TYPE}} work ok function
	if theLoad A_grp = {{G}} then return true
	return false
end

begin {{SYS}}.{{TYPE}} load search ok function
	if theLoad A_grp = {{G}} then return true
	return false
end

begin {{SYS}}.{{TYPE}} park ok function
{{PARK_OK}}
	return false
end
