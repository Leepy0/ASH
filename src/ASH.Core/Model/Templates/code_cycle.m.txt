/* cycle group {{G}} = {{TYPE}}: no calls and no standby CP. One load per active vehicle boards it on its start CP and stays on board for the run.
   Each lap loads at the next load CP, unloads at the next station (the stations take turns, one per lap; A_m = the station's material) and runs
   the return legs. A lap is one job: busy time, time split and lead times count from its start. Swap at a station: from one vehicle leaving it
   to the next one's arrival at its wait point (the buffer, else the unload CP); 0 s when the next one is held there already */
begin P_cycle_{{G}} arriving procedure
	set A_grp to {{G}}
	if A_k = 0 then begin
		/* the start load (model initialization): clone k boards the vehicle on start CP k. V_i is free here: no wait inside the loop */
		set V_i to 1
		while V_i <= V_active({{G}}) do begin
			set A_k to V_i
			clone 1 load to P_cycle_{{G}}
			inc V_i by 1
		end
		send to die
	end
	/* empty: AutoMod cannot hide a load, so it is shrunk out of sight until the pickup */
	scale to x 0.001, y 0.001, z 0.001
	set A_qty to {{CAPACITY}}
{{START}}
	set A_veh to this load vehicle
	set A_vid to A_veh A_index + {{VID_OFFSET}}
	/* the lap; nothing is retrieved: the vehicle sets off from the last unload or return CP */
	while 1 = 1 do begin
		set A_tCall to ac
		set A_tPick to ac
		set A_ret to 0
		set A_hold to 0
		set A_bw to 0
		set A_ld to 0
		set A_dn to 0
		set A_jf to 0
		set A_js0 to V_vT(A_vid + {{NV}})
		set A_jt0 to V_vT(A_vid + {{NV2}})
{{EMPTY_SPEED}}
{{PICK_TRAVEL}}
{{FAIL_LEG}}
{{LOADED_SPEED}}
		/* the next station, and its pickup counts */
		inc V_toRot({{G}}) by 1
		if V_toRot({{G}}) > {{NS}} then set V_toRot({{G}}) to 1
{{TURN}}
		scale to x 1, y 1, z 1
		wait for {{LOAD_TIME}} sec
{{SUPPLY_LEGS}}
		set V_extMat to A_m
		set V_tmp to F_ext_before_unload()
		wait for {{UNLOAD_TIME}} sec
		inc V_stock(A_m) by A_qty
		/* leaving the station: a next vehicle held there already takes over at once (0 s), else the swap runs until one arrives */
		if ac >= V_warmup then begin
			if V_waitVeh(A_m) > 0 then begin
				inc V_swapN(A_m) by 1
				inc V_follow(A_m) by 1
				set V_swapOpen(A_m) to 0
			end
			else begin
				set V_lastLeave(A_m) to ac
				set V_swapOpen(A_m) to 1
			end
		end
{{AFTER_UNLOAD}}
		set A_fin to ac
{{BUSY}}
	end
end
