/* material 1 = M1 (group 1) */
/* one consumer cycle per takt: call check, then consume or stay stopped; a stopped cycle re-checks every takt */
begin P_consume_1 arriving procedure
	while 1 = 1 do begin
		/* reorder on the stock position (on hand + containers already called), as many calls as needed at once; one call is one trip and
		   brings a full vehicle, Capacity containers (the clone keeps A_qty). Also call when the position cannot cover one cycle, or a call
		   threshold below the per-cycle need would stop forever */
		while V_stock(1) + V_pending(1) <= V_call(1) + 0.000001 or V_stock(1) + V_pending(1) < V_need(1) - 0.000001 do begin
			set A_qty to 1
			if V_events = 1 and ac >= V_warmup then print ac "\tcall\tM1\t\t" A_qty to V_ev
			inc V_pending(1) by A_qty
			set V_extMat to 1
			set V_tmp to F_ext_on_call()
			clone 1 load to P_supply_1
		end
		if V_stock(1) < V_need(1) - 0.000001 then begin
			if V_stopped(1) = 0 then begin
				set V_stopped(1) to 1
				set V_stopSince(1) to ac
				if V_events = 1 and ac >= V_warmup then print ac "\tstockout\tM1\t\t0" to V_ev
			end
		end
		else begin
			if V_stopped(1) = 1 then begin
				set V_stopped(1) to 0
				if V_stopSince(1) < V_warmup then set V_stopSince(1) to V_warmup
				if ac > V_stopSince(1) then begin
					inc V_stop(1) by ac - V_stopSince(1)
					if V_events = 1 then print ac "\tstockin\tM1\t\t" ac - V_stopSince(1) to V_ev
				end
			end
			dec V_stock(1) by V_need(1)
			if ac >= V_warmup then inc V_consumed(1) by V_need(1)
		end
		if ac >= V_warmup then begin
			inc V_samples(1) by 1
			tabulate V_stock(1) in T_stock_1
			inc V_stockSum(1) by V_stock(1)
			if V_stock(1) < V_minStock(1) then set V_minStock(1) to V_stock(1)
		end
		wait for V_takt(1) sec
	end
end

/* One supply job, start to end, driven by the load: it enters the system at the group's standby CP, where calls queue first come first served
   whether or not a vehicle is there, boards the first free vehicle, rides along every leg with travel to, and brings the vehicle home before it
   dies, so the park list never decides where a vehicle goes. AutoMod's vehicle states no longer mean anything (the load is on board the whole
   job), so busy time, distance and lead times are kept here. */
begin P_supply_1 arriving procedure
	set A_grp to 1
	set load type to L_m1
	/* a call is no container yet: AutoMod cannot hide a load, so it is shrunk out of sight until the pickup CP */
	scale to x 0.001, y 0.001, z 0.001
	set A_tCall to ac
	set V_extMat to 1
	set V_tmp to F_ext_select_vehicle()
	move into agv:S001
	set A_veh to this load vehicle
	set A_vid to A_veh A_index + 0
	set A_tPick to ac
	set A_hold to 0
	set A_bw to 0
	set A_ld to 0
	set A_dn to 0
	set A_jf to 0
	set A_js0 to V_vT(A_vid + 3)
	set A_jt0 to V_vT(A_vid + 6)
	/* a vehicle that came over from elsewhere to take this load: AutoMod counts that run as retrieving; the share since its last pickup is this job's */
	set V_x to (A_veh percent time retrieving) / 100 * ac
	set A_ret to V_x - A_veh A_rLast
	set A_veh A_rLast to V_x


	/* a run outside the legs (to the pickup, home) measured for the time split like a leg */
	set A_d0 to A_veh total distance traveled
	set V_vT(A_vid) to 1
	travel to agv:L001
	set V_vT(A_vid) to 0
	inc A_jf by (A_veh total distance traveled - A_d0) / 1


	if ac >= V_warmup then begin
		tabulate ac - A_tCall in T_pick_1
		if V_events = 1 then print ac "\tpick\tM1\tL001\t" ac - A_tCall to V_ev
	end
	scale to x 1, y 1, z 1
	wait for 10 sec
	set A_t0 to ac
	set V_legStart(1) to ac
	set V_inLeg(1) to 1
	set A_veh to this load vehicle
	set A_mv0 to (A_veh percent time in motion) / 100 * ac
	set A_d0 to A_veh total distance traveled
	set V_vT(A_vid) to 1
	travel to agv:B001
	set V_vT(A_vid) to 0
	set V_inLeg(1) to 0
	inc A_ld by ac - A_t0
	/* A_t0 lives on the load: a return leg and the next supply of the same material may overlap */
	if ac - A_t0 > V_deadlockSec then inc V_deadlock by 1
	/* wait = leg time minus time the vehicle was moving (AM has no per-vehicle blocked-time attribute); includes decel/accel at the stop */
	set A_w to ac - A_t0 - ((A_veh percent time in motion) / 100 * ac - A_mv0)
	if A_w < 0 then set A_w to 0
	/* time split: the leg at free speed (distance / the leg's speed) */
	inc A_jf by (A_veh total distance traveled - A_d0) / 1
	if V_events = 1 and ac >= V_warmup then print ac "\twait\tM1\tB001\t" A_w to V_ev
	if V_debug = 1 then print "ASH: leg m=1 to B001 dur " (ac - A_t0) " wait " A_w to message

	set A_bw to ac
	set A_hold to ac
	while V_stock(1) > V_unload(1) do begin
		wait for 5 sec
	end
	set A_hold to ac - A_hold
	set A_bw to ac - A_bw
	set A_t0 to ac
	set V_legStart(1) to ac
	set V_inLeg(1) to 1
	set A_veh to this load vehicle
	set A_mv0 to (A_veh percent time in motion) / 100 * ac
	set A_d0 to A_veh total distance traveled
	set V_vT(A_vid) to 1
	travel to agv:U001
	set V_vT(A_vid) to 0
	set V_inLeg(1) to 0
	inc A_ld by ac - A_t0
	/* A_t0 lives on the load: a return leg and the next supply of the same material may overlap */
	if ac - A_t0 > V_deadlockSec then inc V_deadlock by 1
	/* wait = leg time minus time the vehicle was moving (AM has no per-vehicle blocked-time attribute); includes decel/accel at the stop */
	set A_w to ac - A_t0 - ((A_veh percent time in motion) / 100 * ac - A_mv0)
	if A_w < 0 then set A_w to 0
	/* time split: the leg at free speed (distance / the leg's speed) */
	inc A_jf by (A_veh total distance traveled - A_d0) / 1
	if V_events = 1 and ac >= V_warmup then print ac "\twait\tM1\tU001\t" A_w to V_ev
	if V_debug = 1 then print "ASH: leg m=1 to U001 dur " (ac - A_t0) " wait " A_w to message
	set V_extMat to 1
	set V_tmp to F_ext_before_unload()
	wait for 10 sec
	inc V_stock(1) by A_qty
	if ac >= V_warmup then begin
		inc V_supplies(1) by A_qty
		tabulate ac - A_tCall in T_deliv_1
		if V_events = 1 then print ac "\tdeliver\tM1\tU001\t" ac - A_tCall to V_ev
	end
	dec V_pending(1) by A_qty

	scale to x 0.001, y 0.001, z 0.001



	/* home first, always: the next job starts from the standby CP */
	/* a run outside the legs (to the pickup, home) measured for the time split like a leg */
	set A_d0 to A_veh total distance traveled
	set V_vT(A_vid) to 1
	travel to agv:S001
	set V_vT(A_vid) to 0
	inc A_jf by (A_veh total distance traveled - A_d0) / 1

	set A_fin to ac

	/* A_hold = time held for the unload threshold (at the buffer, or at the unload CP without one): not busy (the load, unload and extra times set in the input stay busy) */
	if A_tPick >= V_warmup then begin
		inc V_vBusy(A_vid) by A_ret + A_fin - A_tPick - A_hold
		inc V_vJobs(A_vid) by 1
		/* where the busy time went: loaded legs, load/unload/extra waits, buffer hold within / beyond the allowance, repairs held */
		inc V_vLoad(A_vid) by A_ld
		inc V_vHand(A_vid) by 20
		inc V_vBufB(A_vid) by A_bw - A_hold
		inc V_vBufX(A_vid) by A_hold
		inc V_vRep(A_vid) by A_dn
		set V_x to A_veh total distance traveled
		inc V_vDist(A_vid) by V_x - A_veh A_dLast
		inc V_mDist(1) by V_x - A_veh A_dLast
		inc V_mJobs(1) by 1
		/* time split per group (tsplit row): job time, its runs at free speed, standing in them, the fixed waits, turning on the spot */
		inc V_gT(1) by 1
		inc V_gT(2) by A_fin - A_tPick - A_hold
		inc V_gT(3) by A_jf
		inc V_gT(4) by V_vT(A_vid + 3) - A_js0
		inc V_gT(5) by 20
		inc V_gT(6) by V_vT(A_vid + 6) - A_jt0
	end
	set A_veh A_dLast to A_veh total distance traveled
	send to die
end
